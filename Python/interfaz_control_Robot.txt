def actualizarGauges():
    datos = arduino.readline().decode('utf-8').strip()
    partes = datos.split(',')
    if len(partes) == 3:  # AsegÃºrate de que hay exactamente 3 partes
        temperatura = float(partes[0])
        humedad = float(partes[1])
        presion = float(partes[2])
    lblpresion = Label(gaugesFrame, text = presion,bg = "white", fg = "black", font = ("Arial",15))
    lblpresion.place(x = 220, y = 130) 
    lbltemp = Label(gaugesFrame, text = temperatura,bg = "white", fg = "black", font = ("Arial",15))
    lbltemp.place(x = 220, y = 50)
    lblhum = Label(gaugesFrame, text = humedad,bg = "white", fg = "black", font = ("Arial",15))
    lblhum.place(x = 220, y = 90)
